```python
# backend/src/invoicing/models/expense_policy.py
"""费用标准政策表（v1.1 §4.2/§4.3 + §7.2 ✅5 阈值可配）。

维度：租户 × 类别 × 标准项 × 城市档。查不到 city_tier 时回落 "default"。
"""
from sqlalchemy import (Boolean, Column, DateTime, Integer, Numeric, String,
                        UniqueConstraint, func)
from invoicing.db import Base


class ExpensePolicy(Base):
    __tablename__ = "expense_policies"
    id = Column(Integer, primary_key=True)
    tenant_id = Column(String(64), nullable=False, index=True)
    category = Column(String(32), nullable=False, index=True)   # travel/entertainment/...
    item_key = Column(String(48), nullable=False, index=True)   # accommodation/meal_allowance/...
    city_tier = Column(String(32), nullable=False, default="default")  # default/tier1/tier2
    standard = Column(Numeric(18, 2), nullable=False)
    tolerance = Column(Numeric(18, 2), nullable=False, default=0)
    unit = Column(String(16), nullable=False, default="per_day")  # per_day/per_night/per_person
    note = Column(String(256), nullable=True)
    enabled = Column(Boolean, nullable=False, default=True)
    created_at = Column(DateTime, server_default=func.now(), nullable=False)
    updated_at = Column(DateTime, server_default=func.now(), onupdate=func.now(),
                        nullable=False)
    __table_args__ = (UniqueConstraint("tenant_id", "category", "item_key",
                                       "city_tier", name="uq_policy_dim"),)
```

```python
# backend/src/invoicing/workflow/policy_service.py
"""费用标准查询（v1.1 §4.2/§4.3）：tier 未命中回落 default。"""
from sqlalchemy import select
from invoicing.models.expense_policy import ExpensePolicy


def get_policy(db, *, tenant_id: str, category: str, item_key: str,
               city_tier: str | None = None) -> ExpensePolicy | None:
    rows = db.execute(
        select(ExpensePolicy).where(
            ExpensePolicy.tenant_id == tenant_id,
            ExpensePolicy.category == category,
            ExpensePolicy.item_key == item_key,
            ExpensePolicy.city_tier.in_([city_tier or "default", "default"]),
            ExpensePolicy.enabled.is_(True),
        )
    ).scalars().all()
    if not rows:
        return None
    if city_tier:
        exact = [r for r in rows if r.city_tier == city_tier]
        if exact:
            return exact[0]
    return next((r for r in rows if r.city_tier == "default"), None)


def list_policies(db, *, tenant_id: str, category: str | None = None):
    q = select(ExpensePolicy).where(ExpensePolicy.tenant_id == tenant_id)
    if category:
        q = q.where(ExpensePolicy.category == category)
    return db.execute(q.order_by(ExpensePolicy.category, ExpensePolicy.item_key,
                                 ExpensePolicy.city_tier)).scalars().all()
```

```python
# backend/src/invoicing/workflow/policy_seed.py
"""默认政策种子：值全部取自 settings（spec §7.2 ✅5 不硬编码）。

幂等：按 (category, item_key, city_tier) 已存在则跳过。
"""
from decimal import Decimal
from invoicing.config import settings
from invoicing.models.expense_policy import ExpensePolicy
from invoicing.workflow.policy_service import get_policy

_DEFAULTS = (
    # (category, item_key, unit, 取值函数)
    ("travel", "meal_allowance", "per_day",
     lambda: Decimal(str(settings.travel_allowance_daily_standard))),
    ("travel", "accommodation", "per_night", lambda: Decimal("500")),
    ("travel", "local_transport_day", "per_day", lambda: Decimal("200")),
    ("entertainment", "per_head", "per_person", lambda: Decimal("300")),
)


def seed_default_policies(db, *, tenant_id: str = "default") -> int:
    created = 0
    for category, item_key, unit, value_of in _DEFAULTS:
        if get_policy(db, tenant_id=tenant_id, category=category,
                      item_key=item_key, city_tier="default") is not None:
            continue
        db.add(ExpensePolicy(
            tenant_id=tenant_id, category=category, item_key=item_key,
            city_tier="default", standard=value_of(),
            tolerance=Decimal(str(settings.over_threshold_tolerance)),
            unit=unit, note="系统默认（可在 expense_policies 表调整）",
        ))
        created += 1
    db.commit()
    return created
```

```python
# test/test_skill_docs.py
"""Skill 文档契约：frontmatter 完整、工具名存在、两段握手条款在。"""
import re
from pathlib import Path
import pytest

SKILLS = Path(__file__).resolve().parents[1] / "docs" / "skills"
EXPECTED = ["travel-reimbursement", "meal-reimbursement", "transport-reimbursement",
            "invoice-supplement", "over-budget"]


@pytest.mark.parametrize("slug", EXPECTED)
def test_skill_doc_exists_with_frontmatter(slug):
    text = (SKILLS / slug / "SKILL.md").read_text(encoding="utf-8")
    assert text.startswith("---\n")
    fm = text.split("---", 2)[1]
    assert f"name: {slug}" in fm
    for field in ("description", "version", "display_name", "visibility"):
        assert field in fm


@pytest.mark.parametrize("slug", EXPECTED)
def test_skill_doc_teaches_two_phase_handshake(slug):
    text = (SKILLS / slug / "SKILL.md").read_text(encoding="utf-8")
    assert "confirm_execute" in text and "human_ack" in text


@pytest.mark.parametrize("slug", EXPECTED)
def test_skill_doc_tool_names_exist_in_registry(slug):
    """文档里出现的 invoice-ease 工具名必须在 MCP 注册表里（防教错工具）。"""
    import asyncio
    from invoicing.mcp.server import build_server
    names = {t.name for t in asyncio.run(build_server().list_tools())}
    text = (SKILLS / slug / "SKILL.md").read_text(encoding="utf-8")
    mentioned = set(re.findall(r"\b([a-z_]+(?:_proposal)?)\b", text))
    used = {m for m in mentioned if m in names or f"{m}_proposal" in names}
    assert used, "文档未引用任何真实工具名"


def test_advertisement_budget_under_500_tokens():
    """5 个 skill 的 frontmatter description 合计 ≤ 500 token（spec §5.3 渐进式披露）。"""
    total = 0
    for slug in EXPECTED:
        text = (SKILLS / slug / "SKILL.md").read_text(encoding="utf-8")
        fm = text.split("---", 2)[1]
        desc = re.search(r"description_zh:\s*\"?([^\"\n]+)", fm)
        total += len(desc.group(1)) / 1.5  # 中文约 1.5 字/token 粗估
    assert total <= 500
```
