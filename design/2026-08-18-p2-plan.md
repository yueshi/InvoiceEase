```python
def verify_xml_signature(data: bytes) -> tuple[bool, str | None]:
    """验签 XMLDSig。返回 (通过, 失败原因)；无 Signature 节点 → (False, "no-signature")。

    D4：签名无效由调用方标记 XML_SIGNATURE_INVALID 待复核；无签名记警告不拦。
    """
    from lxml import etree
    from signxml import XMLSigner, XMLVerifier  # XMLVerifier 用验签

    try:
        root = etree.fromstring(data)
    except etree.XMLSyntaxError:
        return False, "xml-parse-error"
    sig = root.find(".//{http://www.w3.org/2000/09/xmldsig#}Signature")
    if sig is None:
        return False, "no-signature"
    try:
        XMLVerifier().verify(data, x509_cert=None, ...)  # 实施时按 signxml API 定参数
        return True, None
    except Exception as exc:  # InvalidSignature 等
        return False, type(exc).__name__  # 不含原始 XML，防泄漏票面数据
```

`parse_invoice_xml` 入口调用：解析成功后若有 Signature 且验签失败 → 返回 None 并把错误写入返回通道（与现有解析错误模式一致——实施时按 ParseOutcome 结构接线；无签名仅记录日志）。

    """配置 URL → POST 企微 markdown 格式；HTTP 失败降级不抛异常。"""
    captured = {}

    class _Resp:
        status_code = 200

        def json(self):
            return {"errcode": 0}

    class _Client:
        def post(self, url, json):
            captured["url"] = url
            captured["json"] = json
            return _Resp()
